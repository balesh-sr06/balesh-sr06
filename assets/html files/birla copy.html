#include using namespace std; long the_cargo_ship(long n) { queue> q; unordered_set visited; q.push({0, 1}); visited.insert("0,1"); long steps = 0; while (!q.empty()) { int size = q.size(); while (size--) { auto [pos, speed] = q.front(); q.pop(); if (pos == n) return steps; long newPos = pos + speed; long newSpeed = speed * 2; string key1 = to_string(newPos) + "," + to_string(newSpeed); if (abs(newPos) <= 2 * n && !visited.count(key1)) { visited.insert(key1); q.push({newPos, newSpeed}); } long revSpeed = (speed > 0) ? -1 : 1; string key2 = to_string(pos) + "," + to_string(revSpeed); if (!visited.count(key2)) { visited.insert(key2); q.push({pos, revSpeed}); } } steps++; } return -1; } int main() { long n; cin >> n; cout << the_cargo_ship(n) << endl; return 0; }
